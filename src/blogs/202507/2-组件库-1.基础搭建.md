# 创建一个 Vue 应用

通过 `vite` 或者 `vue` 创建一个应用，并勾选 `typescript`。

```bash
$ pnpm create vite mino-ui
# 或者
$ pnpm create vue mino-ui
```

- [tailwindcss](!notation:underline)

  添加 `tailwindcss`

  ```bash
  $ pnpm i tailwindcss @tailwindcss/vite -D
  ```

- [vue-tsc](!notation:underline)

  `package.json` 的 `build` 命令为 `vue-tsc -b && vite build`

## 构建组件

1. 创建 [components/number-scroller/index.vue](!notation:box)

```vue [!title:src/components/number-scroller/index.vue]
<template>
  <div class="flex h-4 overflow-hidden" :style="{ height: height + 'px' }">
    <template v-for="(digits, index) in [integerDigits, decimalDigits]">
      <div v-if="index === 1 && digits.length" class="text-center" :style="mergedStyle">.</div>
      <div v-for="(item, index) in digits" :key="index" class="text-center" :style="mergedStyle">
        <div
          class="flex flex-col transition-transform ease-in-out animate"
          :style="{
            '--tw-duration': duration + 'ms',
            '--initial-translate': `${item.initial}px`,
            translate: `0 ${item.offset}px`,
          }"
        >
          <span v-for="(_, i) in 10">{{ i }}</span>
        </div>
      </div>
    </template>
  </div>
</template>

<script setup lang="ts">
import { computed, toRefs } from "vue";
import type { CSSProperties } from "vue";

interface Props {
  num?: number | string;
  duration?: number;
  height?: number;
  backStyle?: CSSProperties;
}

interface DigitType {
  digit: number;
  initial: number;
  offset: number;
}

const props = withDefaults(defineProps<Props>(), {
  num: 0,
  duration: 1000,
  height: 16,
  backStyle: () => ({}),
});

const { num, duration, height, backStyle } = toRefs(props);

const mergedStyle = computed<CSSProperties>(() => ({
  height: `${height.value}px`,
  lineHeight: `${height.value}px`,
  ...backStyle.value,
}));

const integerDigits = computed<DigitType[]>((oldValue) => {
  const [integer] = num.value.toString().split(".");
  return patchDigits(integer.split(""), oldValue!);
});

const decimalDigits = computed<DigitType[]>((oldValue) => {
  const [, decimal] = num.value.toString().split(".");
  return patchDigits(decimal?.split(""), oldValue!);
});

function patchDigits(digits: string[], patchedDigits: DigitType[]): DigitType[] {
  if (!digits) return [];
  return digits.map((d, i) => ({
    digit: patchedDigits?.[i]?.digit ?? ~~d,
    initial: patchedDigits?.[i]?.initial ?? ~~d * -1 * height.value,
    offset: patchedDigits?.[i] ? -1 * height.value * (~~d - patchedDigits[i].digit) : 0,
  }));
}
</script>

<style scoped>
.animate {
  animation: translateY var(--tw-duration) ease-in-out forwards;
}

@keyframes translateY {
  0% {
    transform: translateY(0);
  }
  100% {
    transform: translateY(var(--initial-translate));
  }
}
</style>
```

2. 创建 [components/number-scroller/index.ts](!notation:box)

```markdown [!tip:primary]
这里仿照 `element-plus` 为每一个组件绑定一个自己的 `install` 方法，便于实现单个组件全局注册。

并且可以更好的支持 `tree shaking` 。
```

```typescript [!title:components/number-scroller/index.ts]
// 每个组件的入口文件，可以单独 install
import type { App } from "vue";

import Component from "./index.vue";

function install(app: App) {
  app.component("NumberScroller", Component);
}

Component.install = install;

const NumberScroller = Component;

export { NumberScroller };
```

3. 创建 [components/index.ts](!notation:box)

> 主要是提供安装所有组件的函数(调用每个组件的 install 方法)，并导出所有的组件，便于实现全局注册组件库。

```typescript [!title:components/index.ts]
import type { App } from "vue";
import { NumberScroller } from "./number-scroller";

const components = {
  NumberScroller,
};

export function installAll(app: App) {
  Object.values(components).forEach((comp) => {
    if ("install" in comp) {
      comp.install(app);
    } else {
      console.warn(`Component ${comp} has no install method`);
    }
  });
}

export { NumberScroller };
```

## 入口文件

为要打包为一个库，所以需要提供一个入口文件，创建 `src/index.ts` 。同时需要引入 `style.css` 才能生成样式文件。

```typescript [!title:src/index.ts]
import "./style.css";

import { installAll } from "./components";

export default {
  install: installAll,
};

export { NumberScroller } from "./components";
```
