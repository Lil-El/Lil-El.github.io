# 打包配置 [!toc hide]

关于打包，在 `element-plus` 中，它打包的结果的目录结构如下：

```markdown
element-plus
  ├── dist/
  │   ├── index.css
  │   └── index.full.js
  ├── es/
  │   ├── components/
  │   │   ├── alert/
  │   │   │   ├── src/
  │   │   │   ├── style/
  │   │   │   ├── index.d.ts
  │   │   │   └── index.ts
  │   │   ├── index.d.ts
  │   │   └── index.ts
  │   ├── index.d.ts
  │   └── index.ts
  ├── lib/
  │   ├── components/
  │   │   ├── alert/
  │   │   │   ├── src/
  │   │   │   ├── style/
  │   │   │   ├── index.d.ts
  │   │   │   └── index.ts
  │   │   ├── index.d.ts
  │   │   └── index.ts
  │   ├── index.d.ts
  │   └── index.ts
  ├── global.d.ts
  └── package.json
```

我们对它的产物进行解析：

## 模块

- [es：是基于 esm 模块打包的产物](!notation:box:yellow)
- [lib：是基于 commonjs/cjs 模块打包的产物](!notation:box:yellow)
- [dist：是基于 umd 打包的产物](!notation:box:yellow)

## css 代码分割

对于 `element-plus` 的 es 和 cjs 模块中的每个组件，它都进行了 css 代码分割。

这样在使用该组件库的单个组件时，可以只引用单个组件的样式，避免加载全量的样式。

```markdown [!tip:danger]
`vite` 中的 `build.cssCodeSplit` 默认为 `true`，但是在将项目作为 `lib` 库（即：`build.lib`）打包时，`build.cssCodeSplit` 的默认值是 `false`。

所以必须要对 `build.cssCodeSplit` 显式设置；
```

### 引用单个 element-plus 组件

```js [!title:main.js]
import { ElCard } from "element-plus";
import { createApp } from "vue";

import "element-plus/theme-chalk/base.css";
import "element-plus/theme-chalk/el-card.css";

createApp(App).use(ElCard).mount("#app");
```

## 保留目录结构

`cjs`、`es` 模块配置:

  - 保留原始模块结构：`build.rollupOptions.output[].preserveModules = true;`
  - 指定模块根目录，去掉冗余的 `src` 前缀：`build.rollupOptions.output[].preserveModulesRoot = "src";`

> `preserveModulesRoot` 只对 `es`、`cjs` 模块打包生效

## 代码配置

### package.json

对于 `umd` 模块来说，开启 `build.cssCodeSplit` 会导致结果无法生成 `css` 文件；

所以需要将 `es/cjs` 与 `umd` 的打包配置进行单独构建处理。

同时为了避免每次 `vue-tsc` 处理 `ts` 之后 `vite build` 会将目录内容清除掉，所以需要将 `vue-tsc` 命令设置在打包之后。

```json [!title:package.json]
"scripts": {
  "dev": "vite",
  "build": "pnpm run build:es && pnpm run build:umd",
  "build:es": "vite build --mode es  && vue-tsc -b",
  "build:umd": "vite build --mode umd  && vue-tsc -b",
  "preview": "vite preview"
}
```

### vite.config.ts

```ts [!title:vite.config.ts]
import { defineConfig } from "vite";
import vue from "@vitejs/plugin-vue";
import tailwindcss from "@tailwindcss/vite";
import Components from "unplugin-vue-components/vite";

import type { UserConfig } from "vite";

const baseConfig = {
  plugins: [vue(), tailwindcss(), Components()],
};

// es / cjs 配置
const esConfig: UserConfig = {
  build: {
    cssCodeSplit: true,
    lib: {
      entry: "./src/index.ts",
      // name: "MyUI",
      // fileName: (format) => (format === "es" ? `index.mjs` : `index.${format}.js`),
      // formats: ["es", "umd", "cjs"],
    },
    rollupOptions: {
      external: ["vue"],
      output: [
        {
          format: "es",
          preserveModules: true, // 只在es格式下有效
          preserveModulesRoot: "src",
          entryFileNames: "[name].mjs",
          assetFileNames: (assetInfo) => {
            if (assetInfo.names[0] === "style.css") return "base.css";
            return "[name].css";
          },
          dir: "es",
        },
        {
          format: "commonjs", // 和 cjs 等价
          preserveModules: true,
          preserveModulesRoot: "src",
          entryFileNames: "[name].js",
          assetFileNames: (assetInfo) => {
            if (assetInfo.names[0] === "style.css") return "base.css";
            return "[name].css";
          },
          dir: "lib",
        },
      ],
    },
    emptyOutDir: true,
  },
};

// umd 配置
const umdConfig: UserConfig = {
  build: {
    lib: {
      entry: "./src/index.ts",
      name: "MyUI",
      fileName: (format) => (format === "es" ? `index.mjs` : `index.${format}.js`),
      formats: ["umd"],
    },
    rollupOptions: {
      external: ["vue"],
      output: {
        // name: "MyUI", // 必须为UMD格式指定name
        globals: {
          vue: "Vue",
        },
        dir: "dist",
        assetFileNames: "index.[ext]", // 固定CSS文件名
        inlineDynamicImports: false,
        manualChunks: undefined,
      },
    },
    emptyOutDir: true,
  },
};

export default defineConfig(({ mode }) => {
  return {
    ...baseConfig,
    ...(mode === "es" ? esConfig : umdConfig),
  };
});
```
