它会将源码目录中的 `.ts`、`.vue` 文件自动生成 `.dts` 文件

```markdown [!tip:warning]

```

## tsconfig

`es` 和 `cjs` 打包结构要对原有目录结构进行保留，所以要对这两种模块的打包结果进行处理，将其中的 `*.ts`、`*.vue` 等文件进行 `dts` 处理；

```ts [!title:tsconfig.json]
{
  "files": [],
  "references": [
    { "path": "./tsconfig.app.json" },
    { "path": "./tsconfig.node.json" },
    { "path": "./tsconfig.es.json" },
    { "path": "./tsconfig.cjs.json" }
  ]
}
```

```ts [!title:tsconfig.es.json]
{
  "compilerOptions": {
    "tsBuildInfoFile": "./node_modules/.tmp/tsconfig.es.tsbuildinfo",

    /* Linting */
    "strict": true,
    "noUnusedLocals": true,
    "noUnusedParameters": true,
    "erasableSyntaxOnly": true,
    "noFallthroughCasesInSwitch": true,
    "noUncheckedSideEffectImports": true,

    "target": "ESNext",
    "module": "ESNext",
    "moduleResolution": "node",
    "types": ["vite/client"], // 如果使用 Vite

    "outDir": "es", // 输出目录
    "declarationDir": "es",
    "declaration": true, // 生成 .d.ts 文件
    "emitDeclarationOnly": true, // 只生成声明文件
    "rootDir": "src" // 源文件根目录
  },
  "include": ["src/**/*.ts", "src/**/*.vue"],
  "exclude": ["src/App.vue", "src/main.ts"]
}

```

```ts [!title:tsconfig.cjs.ts]
{
  "compilerOptions": {
    "tsBuildInfoFile": "./node_modules/.tmp/tsconfig.cjs.tsbuildinfo",

    /* Linting */
    "strict": true,
    "noUnusedLocals": true,
    "noUnusedParameters": true,
    "erasableSyntaxOnly": true,
    "noFallthroughCasesInSwitch": true,
    "noUncheckedSideEffectImports": true,

    "target": "ESNext",
    "module": "ESNext",
    "moduleResolution": "node",
    "types": ["vite/client"], // 如果使用 Vite

    "outDir": "lib", // 输出目录
    "declarationDir": "lib",
    "declaration": true, // 生成 .d.ts 文件
    "emitDeclarationOnly": true, // 只生成声明文件
    "rootDir": "src" // 源文件根目录
  },
  "include": ["src/**/*.ts", "src/**/*.vue"],
  "exclude": ["src/App.vue", "src/main.ts"]
}

```

```markdown [!tip:warning]
`tsBuildInfoFile`：配置是 `ts构建信息文件` 的生成位置，如果不配置，每次 `vue-tsc` 会将信息文件生成在根目录下
```
